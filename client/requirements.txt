# Hardware-specific dependencies (not needed for simulation mode):
# 
# LED Hardware (Raspberry Pi only):
# pi5neo - WS2812 LED control library (requires spidev)
# Install on Pi: pip install pi5neo
# 
# Audio Hardware:
# PyAudio - real-time audio capture
# Install with: sudo apt install python3-pyaudio
# Or: pip install pyaudio
# 
# librosa - advanced audio analysis
# Install with: pip install librosa soundfile scipy
# 
# Camera and Face Recognition (for decompression mode):
# opencv-python - camera capture and basic face detection
# Install with: pip install opencv-python
# 
# face-recognition - advanced facial recognition (optional)
# Note: Requires dlib which may need compilation on some systems
# Install with: pip install face-recognition
# On Raspberry Pi: sudo apt install libopenblas-dev liblapack-dev && pip install face-recognition
#
# mediapipe - real-time hand and body tracking for gesture-driven visuals (optional)
# Install with: pip install mediapipe
